  -H "Content-Type: application/json" \
  -H "Authorization: Bearer sk-1234" \

    "prompt": "House",
    "negative": "blurry, low quality",
    "cfg": 12,
    "steps": 30,

  }'

  -H "Authorization: Bearer sk-1234" \

  -F "prompt=Anime style" \
